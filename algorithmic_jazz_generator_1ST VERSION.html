class ReLacksJazzVoicings:
    """
    Created by ReLacks: 4-part & 5-part shell chord voicing algorithms
    containing 7th, 9th, and 13th extensions with rootless option.
    """
    def build_jazz_extension(self, root_midi, chord_type='m9'):
        extensions = {
            'm9':    [0, 3, 7, 10, 14],   # 1 - b3 - 5 - b7 - 9
            '13':    [0, 4, 7, 10, 21],   # 1 - 3 - 5 - b7 - 13
            'maj9':  [0, 4, 7, 11, 14],   # 1 - 3 - 5 - 7 - 9
            '7b9':   [0, 4, 10, 13, 20]   # 1 - 3 - b7 - b9 - b13
        }
        return [root_midi + interval for interval in extensions[chord_type]]
